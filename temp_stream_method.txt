    async def stream_quotes(self, symbols: list, callback):
        if not self.enabled:
            return

        async def quote_handler(data):
            try:
                await callback(data)
            except Exception as e:
                logger.error(f'Callback error: {e}')

        for symbol in symbols:
            self.data_stream.subscribe_quotes(quote_handler, symbol)

        logger.info(f'Starting quote stream: {symbols}')

        try:
            await self.data_stream._run_forever()
        except KeyboardInterrupt:
            logger.info('Stopping stream...')
        except Exception as e:
            logger.error(f'Stream failure: {e}')
        finally:
            await self.data_stream.stop_ws()
